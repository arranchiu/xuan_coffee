window.downloadReceiptPNG = function() { const cardElement = document.getElementById('receipt-card-element'); const btn = document.getElementById('download-png-btn'); btn.disabled = true; btn.textContent = '產生高畫質圖卡中...'; const clone = cardElement.cloneNode(true); clone.id = 'receipt-export-node'; clone.style.width = '520px'; clone.style.maxWidth = '520px'; clone.style.padding = '32px 28px'; clone.style.position = 'absolute'; clone.style.left = '-9999px'; clone.style.top = '-9999px'; clone.style.boxSizing = 'border-box'; clone.style.backgroundColor = '#f5f3ef'; document.body.appendChild(clone); html2canvas(clone, { scale: 2, useCORS: true, backgroundColor: '#f5f3ef', width: 520, windowWidth: 520, onclone: (clonedDoc) => { const el = clonedDoc.getElementById('receipt-export-node'); if (el) el.style.height = 'auto'; } }).then(async canvas => { document.body.removeChild(clone); const orderNum = document.getElementById('rc-order-id').textContent; const fileName = `XUAN_COFFEE_訂單_${orderNum}.png`; canvas.toBlob(async (blob) => { if (!blob) { btn.disabled = false; btn.textContent = '下載訂單分享給店長'; showToast('圖卡生成失敗，請重試'); return; } const dataUrl = canvas.toDataURL('image/png'); const isMobile = /Mobi|Android|iPhone|iPad|iPod/i.test(navigator.userAgent) || window.innerWidth < 640; // 電腦 / 網頁版處理邏輯 if (!isMobile) { const link = document.createElement('a'); link.download = fileName; link.href = dataUrl; link.click(); btn.disabled = false; btn.textContent = '下載訂單分享給店長'; // 關閉訂單圖卡彈窗並平滑滾動回首頁頂部 closeReceiptModal(); window.scrollTo({ top: 0, behavior: 'smooth' }); showToast('訂單圖卡已成功下載！'); return; } // 行動裝置原生分享 API 處理邏輯 const file = new File([blob], fileName, { type: 'image/png' }); if (navigator.canShare && navigator.canShare({ files: [file] })) { try { await navigator.share({ files: [file], title: 'XUAN COFFEE 訂單', text: '這是我的 XUAN COFFEE 訂單明細圖卡' }); btn.disabled = false; btn.textContent = '下載訂單分享給店長'; clearCart(); closeReceiptModal(); window.scrollTo({ top: 0, behavior: 'smooth' }); showToast('請在分享選單中選擇「儲存影像」即可存入相簿！'); return; } catch (shareErr) { console.log('Share prompt dismissed or fallback:', shareErr); } } // 行動裝置長按儲存備援彈窗 const targetImg = document.getElementById('longpress-target-img'); if (targetImg) { targetImg.src = dataUrl; document.getElementById('mobile-img-save-modal').classList.remove('hidden'); } const link = document.createElement('a'); link.download = fileName; link.href = dataUrl; link.click(); btn.disabled = false; btn.textContent = '下載訂單分享給店長'; }, 'image/png'); }).catch(err => { const node = document.getElementById('receipt-export-node'); if (node) document.body.removeChild(node); console.error(err); btn.disabled = false; btn.textContent = '下載訂單分享給店長'; showToast('產圖失敗，請重試'); }); };
